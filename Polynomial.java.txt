public class Polynomial
{
    public double[] coe;
    public Polynomial()
    {
        this.coe=new double[1];
        coe[0]=0;
    }
    public Polynomial(double[] coe)
    {
        this.coe=new double[coe.length];
        for (int i=0;i<coe.length;i++)
        {
            this.coe[i]=coe[i];
        }
    }
    public Polynomial add(Polynomial otherPoly)
    {
        int len=Math.max(this.coe.length, otherPoly.coe.length);
        double[] addresult=new double[len];
        for (int i=0;i<this.coe.length;i++)
        {
            addresult[i]+=this.coe[i];
        }
        for (int j=0;j<otherPoly.coe.length;j++)
        {
            addresult[j]+=otherPoly.coe[j];
        }
        return new Polynomial(addresult);
    }
    public double evaluate(double x)
    {
        double result=0;
        for (int i=0;i<this.coe.length;i++)
        {
            result+=this.coe[i]*Math.pow(x,i);
        }
        return result;
    }
    public boolean hasRoot(double x)
    {
        return Math.abs(this.evaluate(x))<1e-9;
    }


}
